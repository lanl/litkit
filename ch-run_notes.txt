###############################################
# Confirm that PyTorch and CUDA are available #
###############################################

module purge
module load charliecloud/0.42  

# --- Option A: use site CUDA module (recommended on managed clusters)

# Load CUDA 12.5 from the module system (fails if the module isn't installed).
module load cuda/12.5.0

# If the module didn’t set CUDA_HOME, derive it by walking up from nvcc (.../bin/nvcc -> ...).
export CUDA_HOME="${CUDA_HOME:-$(dirname "$(dirname "$(which nvcc)")")}"

# Pick the correct ARM runtime lib directory under CUDA_HOME.
# Prefer SBSA layout (Grace/Hopper on ARM); fall back to older aarch64 layout.
if [ -d "$CUDA_HOME/targets/sbsa-linux/lib" ]; then
  export CUDA_LIBA="$CUDA_HOME/targets/sbsa-linux/lib"
elif [ -d "$CUDA_HOME/targets/aarch64-linux/lib" ]; then
  export CUDA_LIBA="$CUDA_HOME/targets/aarch64-linux/lib"
else
  # Nothing found -> you’re on the wrong arch or the module is broken.
  echo "ERROR: cannot find CUDA targets/*/lib under $CUDA_HOME" >&2; exit 2
fi

# Top-level CUDA libs (e.g., cuBLAS) are here.
export CUDA_LIBB="$CUDA_HOME/lib64"

# Typical use inside containers:
#   LD_LIBRARY_PATH="$CUDA_LIBA:$CUDA_LIBB"
# Or bind both into the container and set LD_LIBRARY_PATH there.

# --- Option B: use a fixed, known CUDA install on the host 

# Root of your local CUDA 12.5 install (no modules involved).
export CUDA_BASE=/path/to/cuda-12.5-host/cuda-12.5

# ARM runtime libs (SBSA layout) and top-level libs.
export CUDA_LIBA="$CUDA_BASE/targets/sbsa-linux/lib"
export CUDA_LIBB="$CUDA_BASE/lib64"

# Same container usage:
#   LD_LIBRARY_PATH="$CUDA_LIBA:$CUDA_LIBB"

export CDI_SPEC_DIR=/path/to/cdi-grace
export IMG="$(pwd)/sqfs/litkit-v0.3.33-aarch64-lean.sqfs"

# verify that PyTorch and CUDA are available
ch-run \
  --unset-env='*' \
  --cdi-dirs="$CDI_SPEC_DIR" \
  --cdi=nvidia.com/gpu=all \
  --bind "$CUDA_LIBA:$CUDA_LIBA" \
  --bind "$CUDA_LIBB:$CUDA_LIBB" \
  --set-env="LD_LIBRARY_PATH=$CUDA_LIBA:$CUDA_LIBB" \
  "$IMG" -- \
  /root/.local/share/uv/tools/litkit/bin/python - <<'PY'
import torch, sys, os
print("Python:", sys.version.split()[0])
print("Torch:", torch.__version__, "CUDA build:", torch.version.cuda)
print("LD_LIBRARY_PATH:", os.environ.get("LD_LIBRARY_PATH"))
print("CUDA available:", torch.cuda.is_available())
if torch.cuda.is_available():
    print("Device 0:", torch.cuda.get_device_name(0),
          "CC:", torch.cuda.get_device_capability(0))
PY
